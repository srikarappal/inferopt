# Serving environment. Install into ONE venv and run everything with that
# interpreter -- the most common failure on a new host is running run.py with a
# different python than the one vLLM lives in, which surfaces as
# `FileNotFoundError: 'vllm'` from deep inside a subprocess.
#
#   python -m pip install -r requirements.txt
#   python -c "import vllm; print(vllm.__version__)"     # must succeed
#
# vLLM is pinned because it is the measurement instrument. 0.26 removed flags
# that earlier versions accepted (--disable-log-requests, --swap-space,
# --cuda-graph-sizes, --tokenizer-pool-size) and renamed Prometheus metrics
# (gpu_cache_usage_perc -> kv_cache_usage_perc). Numbers from a different
# version are not comparable to the ones in docs/.
vllm==0.26.0

# torch is NOT pinned here on purpose. vLLM depends on it and the correct build
# is CUDA-version specific: cu126/cu128 for H100 (sm90), cu130 for GB10 (sm121).
# Let vLLM's own dependency resolution pick it, or install the matching wheel
# from pytorch.org FIRST if this host needs a specific CUDA. The reference
# environment for docs/ was torch 2.11.0+cu130.

transformers==5.14.1
